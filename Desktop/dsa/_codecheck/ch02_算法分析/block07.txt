// 在升序数组 a 中查找 x,返回下标;找不到返回 -1
int binarySearch(const vector<int>& a, int x) {
    int low = 0, high = int(a.size()) - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;  // 防溢出的写法,优于 (low+high)/2
        if (a[mid] < x)      low = mid + 1;
        else if (a[mid] > x) high = mid - 1;
        else return mid;
    }
    return -1;
}
