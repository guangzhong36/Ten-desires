#include <iostream>
#include <chrono>
using namespace std;
using Clock = chrono::steady_clock;   // 单调时钟:不受系统改时间影响

int main() {
    auto t0 = Clock::now();
    // ---- 被测代码 ----
    long long s = 0;
    for (int i = 0; i < 1000000; ++i) s += i;
    // ----------------
    double ms = chrono::duration<double, milli>(Clock::now() - t0).count();
    cout << "耗时: " << ms << " ms, 结果 = " << s << endl;
    return 0;
}
