// 下滤 a[i];n 为当前堆大小
void siftDown(vector<int>& a, int i, int n) {
    while (true) {
        int largest = i;
        for (int c : {2 * i + 1, 2 * i + 2})      // 左子、右子(0 基)
            if (c < n && a[c] > a[largest])
                largest = c;
        if (largest == i) break;
        swap(a[i], a[largest]);
        i = largest;
    }
}

void heapSort(vector<int>& a) {
    int n = int(a.size());
    for (int i = n / 2 - 1; i >= 0; --i)          // 1. 下滤建堆 O(n)
        siftDown(a, i, n);
    for (int i = n - 1; i > 0; --i) {             // 2. 逐个钉死最大元
        swap(a[0], a[i]);
        siftDown(a, 0, i);
    }
}
