// 计时:3 次取最小,每次用副本,结果读走防优化
double msTime(void (*f)(vector<int>&), vector<int> a, long long& guard) {
    double best = 1e18;
    for (int r = 0; r < 3; ++r) {
        vector<int> b = a;
        auto t0 = Clock::now();
        f(b);
        guard += b.front();
        best = min(best, chrono::duration<double, milli>(Clock::now() - t0).count());
    }
    return best;
}
