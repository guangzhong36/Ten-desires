// 对 a[left..right] 升序排序
void insertionSort(vector<int>& a, int left, int right) {
    for (int i = left + 1; i <= right; ++i) {
        int key = a[i];
        int j = i - 1;
        while (j >= left && a[j] > key) {   // 严格大于才右移:稳定
            a[j + 1] = a[j];
            --j;
        }
        a[j + 1] = key;
    }
}
