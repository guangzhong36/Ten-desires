# rank[root] = 树高的上界(只在根上有效),初始 0
parent = list(range(8))
rank = [0] * 8

def find(x):
    while parent[x] != x:
        x = parent[x]
    return x

def union_by_rank(a, b):
    ra, rb = find(a), find(b)
    if ra == rb:
        return
    if rank[ra] < rank[rb]:    # 浅树挂到深树
        ra, rb = rb, ra
    parent[rb] = ra
    if rank[ra] == rank[rb]:   # 两棵同秩树合并
        rank[ra] += 1          # 新树秩 +1
