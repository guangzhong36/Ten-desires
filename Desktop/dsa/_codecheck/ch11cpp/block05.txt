// lab11.cpp —— 实验 11:动态数组扩容的摊还代价实测(第一段:两种策略)
#include <iostream>
#include <vector>
using namespace std;

// 任务 1:带真实拷贝计数器的倍增动态数组
class DoublingArray {
    int* data_ = nullptr;
    size_t size_ = 0, cap_ = 0;
public:
    long long totalCopies = 0, lastCost = 0, maxCost = 0;
    void push_back(int x) {
        lastCost = 0;
        if (size_ == cap_) {                    // 满:倍增扩容
            size_t newCap = cap_ == 0 ? 1 : cap_ * 2;
            int* p = new int[newCap];
            for (size_t i = 0; i < size_; ++i) p[i] = data_[i];
            lastCost = (long long)size_;        // 本次 push 的拷贝代价
            totalCopies += size_;
            if (lastCost > maxCost) maxCost = lastCost;
            delete[] data_;
            data_ = p;
            cap_ = newCap;
        }
        data_[size_++] = x;
    }
    ~DoublingArray() { delete[] data_; }
};

// 任务 1:每次只多 1 格的"笨扩容"(对照策略)
class LinearArray {
    int* data_ = nullptr;
    size_t size_ = 0, cap_ = 0;
public:
    long long totalCopies = 0, lastCost = 0, maxCost = 0;
    void push_back(int x) {
        lastCost = 0;
        if (size_ == cap_) {                    // 满:只多开 1 格
            int* p = new int[cap_ + 1];
            for (size_t i = 0; i < size_; ++i) p[i] = data_[i];
            lastCost = (long long)size_;
            totalCopies += size_;
            if (lastCost > maxCost) maxCost = lastCost;
            delete[] data_;
            data_ = p;
            ++cap_;
        }
        data_[size_++] = x;
    }
    ~LinearArray() { delete[] data_; }
};
