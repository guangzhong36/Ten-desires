#include <vector>
#include <iostream>
using namespace std;
int main() {
    vector<int> v;                 // 空表
    v.push_back(3);                // 尾部插入:3
    v.push_back(1);                // 3 1
    v.insert(v.begin(), 9);        // 头部插入:9 3 1  (O(n),搬移全部元素)
    v.erase(v.begin() + 1);        // 删除下标 1:9 1  (O(n))
    v[1] = 5;                      // 随机访问 O(1):9 5
    for (int x : v) cout << x << ' ';   // 范围 for 遍历
    cout << "\nsize=" << v.size() << " capacity=" << v.capacity();
    return 0;
}
