// C 与 C++ 相同:取模是"向 0 截断"除法的余数
// 公式:余数 = 被除数 - 商 × 除数,商向 0 取整
#include <iostream>
using namespace std;
int main() {
    cout << -7 % 3 << endl;    // -1:商取 -2,余数 -7 - (-2)*3 = -1
    cout << 7 % -3 << endl;    //  1:商取 -2,余数 7 - (-2)*(-3) = 1
    cout << -7 / 3 << endl;    // -2:除法同样向 0 截断,不是向下取整!
    int r = -7 % 3;             // 惯用法:修正成非负余数
    if (r < 0) r += 3;
    cout << r << endl;         // 2:与 Python 的 -7 % 3 一致了
    return 0;
}
