# 双散列:冲突时以第二个散列函数的值做步长
def hash_str2(s, M):           # 第二散列:常用 M2 - (h mod M2),M2 为素数
    return M - 1 - (hash_str(s, M) % (M - 1))   # 值域 1..M-1,步长不为 0
